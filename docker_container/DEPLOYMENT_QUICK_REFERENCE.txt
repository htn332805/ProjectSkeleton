QUICK START (3 minutes)
══════════════════════════════════════════════════════════════════════════════
  1. mkdir ~/forage-production && cd ~/forage-production
  2. curl -O docker-compose.yml from repo
  3. cat > .env << 'CONF'
     TZ=America/Recife
     FORAGE_BROWSER_ENGINE=scrapling
     FORAGE_SEARCH_PROVIDER=forage
     FORAGE_CACHE_TTL_SEARCH=300
     FORAGE_CACHE_TTL_EXTRACT=60
     CONF
  4. docker-compose up -d
  5. curl http://localhost:3672/health
